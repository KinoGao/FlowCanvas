type Props = { authToken: string; config: PlatformConfigDocument; onChange: (config: PlatformConfigDocument) => void };

export function ModelConfigPanel({ authToken, config, onChange }: Props) {
    const { message, modal } = App.useApp();
    const [providerDraft, setProviderDraft] = useState<PlatformProvider | null>(null);
    const [providerOriginalId, setProviderOriginalId] = useState("");
    const [modelDraft, setModelDraft] = useState<PlatformModel | null>(null);
    const [modelOriginalId, setModelOriginalId] = useState("");
    const [discovered, setDiscovered] = useState<Record<string, string[]>>({});
    const [discovering, setDiscovering] = useState("");

    const updateProvider = (id: string, patch: Partial<PlatformProvider>) => onChange({ ...config, providers: config.providers.map((item) => item.id === id ? { ...item, ...patch } : item) });
    const updateModel = (id: string, patch: Partial<PlatformModel>) => onChange({ ...config, models: config.models.map((item) => item.id === id ? { ...item, ...patch } : item) });
    const editProvider = (item?: PlatformProvider) => { setProviderOriginalId(item?.id || ""); setProviderDraft(item ? { ...item } : emptyProvider()); };
    const editModel = (item?: PlatformModel) => { setModelOriginalId(item?.id || ""); setModelDraft(item ? cloneModel(item) : emptyModel(config.providers[0]?.id || "")); };

    const saveProvider = () => {
        if (!providerDraft) return;
        const item = { ...providerDraft, id: providerDraft.id.trim().toLowerCase(), name: providerDraft.name.trim(), baseUrl: providerDraft.baseUrl.trim(), modelsPath: providerDraft.modelsPath.trim() || "/models" };
        if (!item.id || !item.name) return void message.warning("????? ID ???");
        if (config.providers.some((next) => next.id === item.id && next.id !== providerOriginalId)) return void message.warning("?? ID ???");
        const models = providerOriginalId && providerOriginalId !== item.id ? config.models.map((model) => model.providerId === providerOriginalId ? { ...model, providerId: item.id } : model) : config.models;
        onChange({ ...config, providers: replaceById(config.providers, providerOriginalId, item), models });
        setProviderDraft(null);
    };
    const saveModel = () => {
        if (!modelDraft) return;
        const item = normalizeModel(modelDraft);
        if (!item.id || !item.displayName || !item.requestModel || !item.providerId) return void message.warning("????? ID??????????????");
        if (config.models.some((next) => next.id === item.id && next.id !== modelOriginalId)) return void message.warning("?? ID ???");
        onChange({ ...config, models: replaceById(config.models, modelOriginalId, item) });
        setModelDraft(null);
    };
    const discover = async (provider: PlatformProvider) => {
        setDiscovering(provider.id);
        try {
            const models = await discoverProviderModels(authToken, provider.id);
            setDiscovered((current) => ({ ...current, [provider.id]: models }));
            message.success(models.length ? `?? ${models.length} ?????` : "???????????????????");
        } catch (error) {
            message.error(error instanceof Error ? error.message : "??????");
        } finally { setDiscovering(""); }
    };

    return <Space direction="vertical" size={18} className="w-full">
        <AdminCard title="????" description="??????????????? Base URL?API Key ?????????????" action={<Button type="primary" icon={<Plus className="size-4" />} onClick={() => editProvider()}>????</Button>}>
            <Table rowKey="id" pagination={false} dataSource={config.providers} scroll={{ x: 980 }} columns={[
                { title: "??", key: "name", render: (_, item) => <div><div className="font-medium">{item.name}</div><div className="text-xs text-gray-400">{item.id}</div></div> },
                { title: "??", dataIndex: "apiFormat", width: 120, render: (value) => <Tag>{String(value)}</Tag> },
                { title: "????", dataIndex: "modelsPath", width: 150 },
                { title: "????", key: "ready", width: 120, render: (_, item) => item.enabled && item.baseUrl && item.apiKey ? <Tag color="green">??</Tag> : <Tag color="orange">?????</Tag> },
                { title: "??", dataIndex: "enabled", width: 80, render: (value, item) => <Switch size="small" checked={value} onChange={(enabled) => updateProvider(item.id, { enabled })} /> },
                { title: "??", key: "actions", width: 230, render: (_, item) => <Space><Button icon={<RefreshCw className="size-4" />} loading={discovering === item.id} onClick={() => void discover(item)}>????</Button><Button type="text" title="????" icon={<Pencil className="size-4" />} onClick={() => editProvider(item)} /><Button danger type="text" title="????" icon={<Trash2 className="size-4" />} onClick={() => modal.confirm({ title: "?????", content: "???????????????", okText: "??", okButtonProps: { danger: true }, onOk: () => onChange({ ...config, providers: config.providers.filter((next) => next.id !== item.id), models: config.models.filter((model) => model.providerId !== item.id) }) })} /></Space> },
            ]} />
        </AdminCard>
        <AdminCard title="????" description="?? ID ?????????????????????????????????????????" action={<Button type="primary" icon={<Plus className="size-4" />} disabled={!config.providers.length} onClick={() => editModel()}>????</Button>}>
            <Table rowKey="id" dataSource={config.models} scroll={{ x: 1120 }} pagination={{ pageSize: 12, hideOnSinglePage: true }} columns={[
                { title: "????", key: "name", render: (_, item) => <div><div className="font-medium">{item.displayName}</div><div className="text-xs text-gray-400">{item.id}</div></div> },
                { title: "??", dataIndex: "providerId", width: 130 },
                { title: "??????", dataIndex: "requestModel", width: 220 },
                { title: "??", dataIndex: "capability", width: 90, render: (value) => <Tag color="blue">{String(value)}</Tag> },
                { title: "??", dataIndex: "modes", width: 220, render: (values: string[]) => <Space size={[4, 4]} wrap>{values.map((value) => <Tag key={value}>{value}</Tag>)}</Space> },
                { title: "????", key: "visible", width: 110, render: (_, item) => <Switch size="small" checked={item.enabled && item.published} onChange={(enabled) => updateModel(item.id, { enabled, published: enabled })} /> },
                { title: "??", key: "actions", width: 100, render: (_, item) => <Space><Button type="text" icon={<Pencil className="size-4" />} onClick={() => editModel(item)} /><Button danger type="text" icon={<Trash2 className="size-4" />} onClick={() => onChange({ ...config, models: config.models.filter((next) => next.id !== item.id) })} /></Space> },
            ]} />
        </AdminCard>
        <ProviderDrawer draft={providerDraft} onChange={setProviderDraft} onClose={() => setProviderDraft(null)} onSave={saveProvider} />
        <ModelDrawer draft={modelDraft} providers={config.providers} discovered={discovered} onChange={setModelDraft} onClose={() => setModelDraft(null)} onSave={saveModel} />
    </Space>;
}

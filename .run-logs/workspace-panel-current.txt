import { App, Button, Descriptions, Drawer, Empty, Form, Input, Modal, Select, Space, Table, Tag } from "antd";
import { Eye, HardDrive, Image, KeyRound, PanelsTopLeft, Pencil, Trash2, Users } from "lucide-react";
import { useState } from "react";

import { AdminCard } from "./admin-card";
import {
    deleteAdminProject, deleteAdminUser, fetchAdminProject, resetAdminUserPassword, updateAdminUser,
    type AdminProjectDetail, type AdminProjectSummary, type AdminUserWorkspace,
} from "@/services/api/platform-admin";

type Props = { authToken: string; users: AdminUserWorkspace[]; onRefresh: () => Promise<void> | void };

export function WorkspacePanel(props: Props) {
    const { message, modal } = App.useApp();
    const [detail, setDetail] = useState<AdminProjectDetail | null>(null);
    const [loadingProject, setLoadingProject] = useState("");
    const [editing, setEditing] = useState<AdminUserWorkspace | null>(null);
    const [passwordUser, setPasswordUser] = useState<AdminUserWorkspace | null>(null);
    const [newPassword, setNewPassword] = useState("");

    const inspect = async (userId: string, project: AdminProjectSummary) => {
        setLoadingProject(userId + ":" + project.id);
        try { setDetail(await fetchAdminProject(props.authToken, userId, project.id)); }
        catch (error) { message.error(error instanceof Error ? error.message : "??????"); }
        finally { setLoadingProject(""); }
    };
    const saveUser = async () => {
        if (!editing) return;
        try {
            await updateAdminUser(props.authToken, editing.id, { username: editing.username.trim(), displayName: editing.displayName.trim(), role: editing.role });
            message.success("?????"); setEditing(null); await props.onRefresh();
        } catch (error) { message.error(error instanceof Error ? error.message : "??????"); }
    };
    const resetPassword = async () => {
        if (!passwordUser || newPassword.length < 6) return void message.warning("???? 6 ?");
        try {
            await resetAdminUserPassword(props.authToken, passwordUser.id, newPassword);
            message.success("??????????????"); setPasswordUser(null); setNewPassword("");
        } catch (error) { message.error(error instanceof Error ? error.message : "??????"); }
    };
    const removeUser = (user: AdminUserWorkspace) => modal.confirm({
        title: `???? ${user.username}?`, content: "???????????????????????????", okText: "????", okButtonProps: { danger: true },
        onOk: async () => { await deleteAdminUser(props.authToken, user.id); message.success("?????"); await props.onRefresh(); },
    });
    const removeProject = (user: AdminUserWorkspace, project: AdminProjectSummary) => modal.confirm({
        title: `???? ${project.title || project.id}?`, content: "????????????????", okText: "????", okButtonProps: { danger: true },
        onOk: async () => { await deleteAdminProject(props.authToken, user.id, project.id); message.success("?????"); await props.onRefresh(); },
    });
    const totals = props.users.reduce((result, user) => ({ projects: result.projects + user.activeProjectCount, assets: result.assets + user.assetCount, files: result.files + user.fileCount, bytes: result.bytes + user.fileBytes }), { projects: 0, assets: 0, files: 0, bytes: 0 });

    return <Space direction="vertical" size={18} className="w-full">
        <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4"><Metric icon={<Users />} label="??" value={props.users.length} /><Metric icon={<PanelsTopLeft />} label="????" value={totals.projects} /><Metric icon={<Image />} label="??" value={totals.assets} /><Metric icon={<HardDrive />} label="????" value={totals.files} detail={formatBytes(totals.bytes)} /></div>
        <AdminCard title="?????" description="??????????????????????????????????????">
            <Table rowKey="id" dataSource={props.users} scroll={{ x: 1120 }} pagination={{ pageSize: 15, hideOnSinglePage: true }} expandable={{ rowExpandable: (user) => user.projects.length > 0, expandedRowRender: (user) => <ProjectTable user={user} loadingProject={loadingProject} onInspect={inspect} onDelete={removeProject} /> }} columns={[
                { title: "??", key: "account", render: (_, user) => <div><div className="font-medium">{user.displayName || user.username}</div><div className="text-xs text-gray-400">{user.username} ? {user.id}</div></div> },
                { title: "??", dataIndex: "role", width: 100, render: (role) => <Tag color={role === "ADMIN" ? "gold" : "default"}>{role === "ADMIN" ? "???" : "??"}</Tag> },
                { title: "??", key: "projects", width: 120, render: (_, user) => <span>{user.activeProjectCount} / {user.projectCount}</span> },
                { title: "??", dataIndex: "assetCount", width: 90 },
                { title: "????", key: "files", width: 150, render: (_, user) => <span>{user.fileCount} ? ? {formatBytes(user.fileBytes)}</span> },
                { title: "????", dataIndex: "updatedAt", width: 180, render: formatDate },
                { title: "??", key: "actions", width: 150, render: (_, user) => <Space><Button type="text" title="????" icon={<Pencil className="size-4" />} onClick={() => setEditing({ ...user })} /><Button type="text" title="????" icon={<KeyRound className="size-4" />} onClick={() => { setPasswordUser(user); setNewPassword(""); }} /><Button danger type="text" title="????" icon={<Trash2 className="size-4" />} onClick={() => removeUser(user)} /></Space> },
            ]} />
        </AdminCard>
        <Drawer title={detail?.title || "????"} width={760} open={Boolean(detail)} onClose={() => setDetail(null)}>{detail ? <><Descriptions size="small" column={2} className="mb-5" items={[{ key: "id", label: "?? ID", children: detail.id }, { key: "user", label: "?? ID", children: detail.userId }, { key: "updated", label: "????", children: formatDate(detail.updatedAt) }, { key: "status", label: "??", children: detail.deletedAt ? <Tag>???</Tag> : <Tag color="green">??</Tag> }]} /><div className="mb-2 text-sm font-medium">??????</div><pre className="max-h-[calc(100vh-220px)] overflow-auto rounded-2xl bg-[#111113] p-4 text-xs leading-6 text-white/80">{JSON.stringify(detail.project, null, 2)}</pre></> : null}</Drawer>
        <Modal title="????" open={Boolean(editing)} onCancel={() => setEditing(null)} onOk={() => void saveUser()} okText="??">{editing ? <Form layout="vertical"><Form.Item label="???"><Input value={editing.username} onChange={(event) => setEditing({ ...editing, username: event.target.value })} /></Form.Item><Form.Item label="????"><Input value={editing.displayName} onChange={(event) => setEditing({ ...editing, displayName: event.target.value })} /></Form.Item><Form.Item label="??"><Select value={editing.role} options={[{ value: "USER", label: "????" }, { value: "ADMIN", label: "???" }]} onChange={(role) => setEditing({ ...editing, role })} /></Form.Item></Form> : null}</Modal>
        <Modal title={`?? ${passwordUser?.username || "??"} ???`} open={Boolean(passwordUser)} onCancel={() => setPasswordUser(null)} onOk={() => void resetPassword()} okText="????"><Input.Password autoFocus placeholder="?? 6 ???" value={newPassword} onChange={(event) => setNewPassword(event.target.value)} onPressEnter={() => void resetPassword()} /></Modal>
    </Space>;
}

function ProjectTable(props: { user: AdminUserWorkspace; loadingProject: string; onInspect: (userId: string, project: AdminProjectSummary) => void; onDelete: (user: AdminUserWorkspace, project: AdminProjectSummary) => void }) {
    if (!props.user.projects.length) return <Empty image={Empty.PRESENTED_IMAGE_SIMPLE} description="????" />;
    return <Table rowKey="id" size="small" pagination={false} dataSource={props.user.projects} columns={[
        { title: "??", key: "title", render: (_, project) => <div><div>{project.title || "?????"}</div><div className="text-xs text-gray-400">{project.id}</div></div> },
        { title: "??", key: "status", width: 100, render: (_, project) => project.deletedAt ? <Tag>???</Tag> : <Tag color="green">??</Tag> },
        { title: "????", dataIndex: "updatedAt", width: 180, render: formatDate },
        { title: "??", key: "actions", width: 110, render: (_, project) => <Space><Button type="text" title="??????" icon={<Eye className="size-4" />} loading={props.loadingProject === props.user.id + ":" + project.id} onClick={() => void props.onInspect(props.user.id, project)} /><Button danger type="text" title="????" icon={<Trash2 className="size-4" />} onClick={() => props.onDelete(props.user, project)} /></Space> },
    ]} />;
}

function Metric(props: { icon: React.ReactNode; label: string; value: number; detail?: string }) { return <div className="flex items-center gap-3 rounded-[22px] border border-white/70 bg-white/86 px-4 py-4 shadow-[0_12px_35px_rgba(0,0,0,0.06)] dark:border-white/10 dark:bg-white/8"><div className="flex size-10 items-center justify-center rounded-full bg-black/[0.05] text-gray-600 dark:bg-white/10 dark:text-white/70 [&_svg]:size-4">{props.icon}</div><div><div className="text-xs text-gray-400">{props.label}</div><div className="text-xl font-semibold">{props.value}<span className="ml-2 text-xs font-normal text-gray-400">{props.detail}</span></div></div></div>; }
function formatDate(value: string) { if (!value) return "-"; const date = new Date(value); return Number.isNaN(date.getTime()) ? value : date.toLocaleString("zh-CN", { hour12: false }); }
function formatBytes(value: number) { if (!value) return "0 B"; const units = ["B", "KB", "MB", "GB", "TB"]; const index = Math.min(Math.floor(Math.log(value) / Math.log(1024)), units.length - 1); return (value / 1024 ** index).toFixed(index ? 1 : 0) + " " + units[index]; }
